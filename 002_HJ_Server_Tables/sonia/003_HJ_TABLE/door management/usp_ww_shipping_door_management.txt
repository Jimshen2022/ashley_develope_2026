 DECLARE     @in_vchLoc1  VARCHAR(50) = 'D100'  
      ,@in_vchLoc2 VARCHAR(50) = 'D600'  
      ,@in_vchLoc3 VARCHAR(50) = '%'  
      ,@in_vchLoc4 VARCHAR(50) = '%'  
      ,@in_vchLoc5 VARCHAR(50) = '%'  
      ,@in_vchLoc6 VARCHAR(50) = '%'  
      ,@in_vchLoc7 VARCHAR(50) = '%'  
      ,@in_vchLoc8 VARCHAR(50) = '%'  
      ,@in_vchLoc9 VARCHAR(50) = '%'

CREATE PROCEDURE [dbo].[usp_ww_shipping_door_management]  
      @in_vchLoc1  VARCHAR(50),  
      @in_vchLoc2  VARCHAR(50),  
      @in_vchLoc3  VARCHAR(50),  
      @in_vchLoc4  VARCHAR(50),  
      @in_vchLoc5  VARCHAR(50),  
      @in_vchLoc6  VARCHAR(50),  
      @in_vchLoc7  VARCHAR(50),  
      @in_vchLoc8  VARCHAR(50),  
      @in_vchLoc9  VARCHAR(50),  
      @in_vchLoc10 VARCHAR(50),  
      @in_vchLoc11 VARCHAR(50),  
      @in_vchLoc12 VARCHAR(50),  
      @in_vchArea  VARCHAR(5)  
AS  
  
SET NOCOUNT ON  
  
IF OBJECT_ID('tempdb..#report') IS NOT NULL DROP TABLE #report  
CREATE TABLE #report (  
 Color     VARCHAR(30),  
 StagingCapacity   DECIMAL(15,2),  
 StagingCubes   DECIMAL(15,2),  
 ID      VARCHAR(5),  
 Door     VARCHAR(50),  
 LoadMinRemain   INT, -- 09/14/2010 Kevin Conan - Added for loading minutes remaining  
 equipment_id   VARCHAR(50),  
 LoadID     VARCHAR(30),  
 nbr_of_orders   VARCHAR(20),  
 pct_loaded    INT,  
 pct_picked    INT,  
 UPH_Need    INT,  
 UPH_Upper    INT, -- 09/08/2010 Kevin Conan - Added UPH Upper  
 upholstery_stage_loc VARCHAR(50),  
 Non_Fwd_Bld_Qty   INT, --10/05/2010 Kevin Conan - Added Non Fwd Pick Building Quantity  
 CG_Need     INT, --09/09/2010 Kevin Conan - Added CG Ground  
 UPPER_Need    INT,  
 PCS_to_load    INT,  
 StagingPctFull   INT,  
 disp_datetime   VARCHAR(35),  
 [status]    VARCHAR(5),  
 current_trailer_loc  VARCHAR(50),  
 zone     VARCHAR(10),  
 loader_link    VARCHAR(30),  
 wh_id     VARCHAR(10),  
 area_id     INT, --02/21/2011 Kevin Conan - Added area_id for the YA table queries  
 due      DATETIME,  
 work_q_id    INT,  
 num_users    INT, --11/02/2010 Angi Brown - Added number users assigned  
 Mload_id    varchar(30) -----------add by grace liu 1/23/2011       
)  
  
  
BEGIN  
      INSERT INTO #report  
   SELECT  0 AS Color,  
     null AS StagingCapacity,  
     null AS StagingCubes,  
     tempselect.ID,   
     tempselect.Door AS Door,   
     (CASE WHEN lms.upholstery_stage_loc IS NULL THEN NULL   
     ELSE (SELECT ROUND(SUM(sto.actual_qty/rate.load_rate_per_hour)*60,0)  
        FROM t_stored_item sto (nolock)   
        JOIN v_item_uom_factored_cube itu (nolock)   
       ON itu.item_number = sto.item_number   
       and itu.wh_id=sto.wh_id -----------9-29-2014 SChen task add muti wh_id for wanek  
        JOIN t_rates rate (nolock)   
       ON rate.pick_area = itu.default_pick_area   
       WHERE sto.location_id = lms.upholstery_stage_loc )  
     END) AS LoadMinRemain,  
     tempselect.equipment_id,   
     (CASE WHEN lms.load_type NOT IN ('B')   
        THEN lms.load_id + '     (transfer)'   
        ELSE lms.load_id   
      END) AS LoadID,  
     --(SELECT  MAX(RIGHT(order_number,2))   
     -- FROM t_order (nolock)   
     --   WHERE LEFT(load_id,8) = LEFT(lms.load_id,8))  AS nbr_of_orders,  
     (SELECT  CASE WHEN lms.load_type IN ('B') then  MAX(distinct orb.drop_number) else 0 end  /***OMS change****/   
      FROM t_order o (nolock)   
      JOIN t_order_detail_breakdown orb  
      ON o.order_number = orb.order_number  
        WHERE o.load_id = lms.load_id  
        )  AS nbr_of_orders,/** this is mapped to number of drops in webwise**/  
     alv.pct_loaded,   
     null AS pct_picked,  
     null AS UPH_Need,  
     null AS UPH_Upper,  
     (CASE WHEN lms.load_type NOT IN ('B')   
        THEN NULL   
        ELSE lms.upholstery_stage_loc   
       END) AS upholstery_stage_loc,   
     null AS Non_Fwd_Bld_Qty,  
     null AS CG_Need,  
     null AS UPPER_Need,  
     null AS PCS_to_load,  
     null AS StagingPctFull,   
     (CONVERT(VARCHAR(5),lms.dispatch_date,1) + ' '+ CONVERT(VARCHAR(5),lms.dispatch_time,108)) AS disp_datetime,   
     lms.status,   
     L2.location_name AS current_trailer_loc,   
     tempselect.zone,  
     tempselect.loader_link,  
     lms.wh_id,  
     tempselect.area_id,  
     tempselect.due,  
     tempselect.work_q_id,  
     0 AS num_users,  
     lms.load_id AS Mload_id  
   FROM (SELECT   '1' AS ID,   
         L.location_name AS Door,   
         ' ' AS zone,   
         'Loaders' AS loader_link,   
         L.area_id,           
         '' AS due ,  
         NULL AS work_q_id,  
         T.trailer_id,  
         T.equipment_id,  
         T.location_id 
      FROM  t_ya_location L(NOLOCK)    
     LEFT JOIN t_trailer T(NOLOCK)   
         ON L.location_id = T.location_id   
         and L.area_id=T.area_id   
        AND T.status not in ('HISTORY','LOST')      
      WHERE L.area_id LIKE @in_vchArea   
      AND (L.location_name BETWEEN @in_vchLoc1 AND @in_vchLoc2   
        OR L.location_name BETWEEN @in_vchLoc3 AND @in_vchLoc4  
        OR L.location_name BETWEEN @in_vchLoc5 AND @in_vchLoc6  
        OR L.location_name BETWEEN @in_vchLoc7 AND @in_vchLoc8  
        OR L.location_name BETWEEN @in_vchLoc9 AND @in_vchLoc10  
        OR L.location_name BETWEEN @in_vchLoc11 AND @in_vchLoc12)  
    UNION  
    SELECT  '2' AS ID,   
       L.location_name AS Door,  
       L.location_name AS zone,   
       'Loaders' AS loader_link,   
        L.area_id,  
       Q.due,  
       Q.work_q_id,  
       Q.trailer_id,  
       T.equipment_id,  
       T.location_id  
     FROM t_ya_work_q Q (nolock)   
     JOIN t_ya_location L (nolock)   
       ON L.location_name = Q.zone   
       and Q.area_id=L.area_id   
     JOIN t_trailer T (nolock)   
       ON T.trailer_id = Q.trailer_id   
       and Q.area_id=T.area_id  
      AND T.status NOT IN ('HISTORY','LOST','IN DOOR')       
       WHERE Q.status <> 'COMPLETE'   
      AND Q.type = '52'   
      AND L.type = 'DOOR'   
      AND L.area_id  LIKE @in_vchArea   
     AND (L.location_name BETWEEN @in_vchLoc1 AND @in_vchLoc2   
       OR L.location_name BETWEEN @in_vchLoc3 AND @in_vchLoc4  
       OR L.location_name BETWEEN @in_vchLoc5 AND @in_vchLoc6  
       OR L.location_name BETWEEN @in_vchLoc7 AND @in_vchLoc8  
       OR L.location_name BETWEEN @in_vchLoc9 AND @in_vchLoc10  
       OR L.location_name BETWEEN @in_vchLoc11 AND @in_vchLoc12) ) tempselect  
    LEFT JOIN t_load_master lms (nolock)   
    ON tempselect.equipment_id = lms.equipment_id  
    AND lms.status NOT IN ('S','Q')   
    LEFT JOIN t_afo_load_view alv (nolock)   
     ON alv.wh_id = lms.wh_id   
    AND alv.load_id = lms.load_id     
    LEFT JOIN t_ya_location L2 (nolock)   
     ON tempselect.location_id = L2.location_id    
    
  ----------------------insert into temp table #tempsum for calculate fields' value  
  IF OBJECT_ID('tempdb..#tempsum','U') IS NOT NULL DROP TABLE #tempsum  
  
  SELECT pkd.load_id,  
    pkd.wh_id,  
    SUM(CASE WHEN pkd.pick_area = 'UPHOLSTERY'   
        AND tier.tier = 1   
       THEN pkd.planned_quantity-pkd.picked_quantity   
          ELSE null   
     END) AS temp11_qty,  
    SUM(CASE WHEN pkd.pick_area = 'UPHOLSTERY'   
        AND tier.tier > 1   
       THEN pkd.planned_quantity-pkd.picked_quantity   
       ELSE null   
     END) AS temp14_qty,  
    SUM(CASE WHEN pkd.pick_area = 'CASEGOOD'     
       THEN pkd.planned_quantity-pkd.picked_quantity   
       ELSE null   
     END) AS temp15_qty,  
    SUM(CASE WHEN pkd.pick_area = 'CASEUPPER'    
       THEN pkd.planned_quantity-pkd.picked_quantity   
       ELSE null   
     END) AS temp12_qty,  
    SUM(CASE WHEN tc.c1 IS NOT NULL    
       THEN pkd.planned_quantity-pkd.picked_quantity   
       ELSE null   
     END)  AS temp16_qty  
  INTO #tempsum  
  FROM t_pick_detail pkd (nolock)  
  JOIN #report r   
    --on pkd.load_id = left(r.LoadID,10) /***OMS change****/  
    ON pkd.load_id = r.LoadID  
   AND pkd.wh_id=r.wh_id   
   AND r.LoadID is not null  
  JOIN t_location loc (nolock)   
    ON loc.location_id = pkd.pick_location   
    and pkd.wh_id=loc.wh_id   
  JOIN v_location_aisle_section_tier tier (nolock)   
    ON tier.location_id = loc.location_id   
    and tier.wh_id=pkd.wh_id   
   LEFT JOIN t_control tc (nolock)   
    ON tc.control_type = 'BLDG_OVERFLOW'   
   AND tc.c1 = tier.building      
    WHERE pkd.user_assigned IS NULL   
      AND pkd.status not in ('HOLD', 'PICKED')  
 GROUP BY pkd.load_id,pkd.wh_id  
  
    
  UPDATE #report  
     SET PCS_to_load  = ( SELECT m.netqty  
        FROM( SELECT orm.load_id /*LEFT(odb.order_number,10)*/ AS order_number,  
            SUM(odb.qty) AS netqty  
             FROM t_order_detail_breakdown odb(nolock)   
          JOIN dbo.t_order orm(nolock) on odb.wh_id=orm.wh_id and odb.order_number=orm.order_number  
            WHERE ISNULL(odb.ship_status, '') NOT IN('BACKORDER', 'B', 'LOADED')  
         GROUP BY orm.load_id ) m   
          WHERE m.order_number = Mload_id   
         AND Mload_id is not null) ,  
     StagingCubes = (SELECT SUM(ISNULL(itu.factored_cube, 0) * ( CASE  WHEN pkd.status IN ( 'PICKED', 'STAGED' )   
                     THEN ( pkd.picked_quantity - pkd.loaded_quantity )  
                     ELSE ( pkd.planned_quantity - pkd.loaded_quantity )  
                   END ))   
        FROM t_pick_detail pkd (nolock)   
        INNER JOIN v_item_uom_factored_cube itu (nolock)   
          ON itu.wh_id = pkd.wh_id   
         AND itu.item_number = pkd.item_number   
         AND itu.uom = pkd.uom   
             WHERE (pkd.user_assigned IS NOT NULL   
           OR pkd.status IN ( 'PICKED', 'STAGED' ))  
          AND pkd.load_id = Mload_id   
          AND Mload_id is not null),  
        UPH_Need =  ( SELECT m.temp11_qty  
          FROM #tempsum m   
         WHERE m.load_id = Mload_id   
        AND Mload_id is not null ),  
       UPH_Upper =  ( SELECT m.temp14_qty  
          FROM #tempsum m   
         WHERE m.load_id = Mload_id   
          AND Mload_id is not null ),  
    Non_Fwd_Bld_Qty =  ( SELECT m.temp16_qty  
         FROM #tempsum m   
        WHERE m.load_id = Mload_id    
          AND Mload_id is not null),  
    CG_Need  = ( SELECT m.temp15_qty  
        FROM #tempsum m   
          WHERE m.load_id = Mload_id   
            AND Mload_id is not null),  
     UPPER_Need = ( SELECT m.temp12_qty  
          FROM #tempsum m   
            WHERE m.load_id = Mload_id   
              AND Mload_id is not null ),        
   StagingCapacity = ( SELECT TOP 1 loc.capacity_volume  
         FROM t_pick_detail pkd (nolock)  
         INNER JOIN t_location loc (nolock)   
           ON loc.location_id = pkd.staging_location   
          AND loc.wh_id = pkd.wh_id  
              WHERE pkd.load_id = Mload_id   
          AND Mload_id is not null),  
    pct_picked = ( SELECT ROUND(SUM(pkd.picked_quantity)/ SUM(pkd.planned_quantity) *100,0)  
         FROM t_pick_detail pkd (nolock)  
        WHERE pkd.load_id=Mload_id   
          AND Mload_id is not null),  
     num_users = ( SELECT COUNT(DISTINCT pkd.user_assigned)  
         FROM t_pick_detail pkd (nolock)  
        WHERE pkd.load_id = Mload_id   
          AND Mload_id is not null),  
    disp_datetime = ( CASE WHEN EXISTS(SELECT 'TRUE'   
            FROM t_work_q wkq (NOLOCK)  
            JOIN t_work_types wks (NOLOCK)  
              ON wkq.work_type = wks.work_type  
              and wkq.wh_id=wks.wh_id  
              WHERE wkq.pick_ref_number = Mload_id   
             AND Mload_id is not null  
            AND wkq.priority > wks.default_priority )  
         THEN '<b><i>' + disp_datetime + '</i></b>'  
         ELSE disp_datetime  
          END)  
   WHERE Mload_id is not null   
  
 END  
  
  
update #report  
 set StagingPctFull = case when StagingCapacity =0 then ISNULL((SELECT 0  
                 FROM t_load_master ldm (NOLOCK)   
                 WHERE ldm.load_id = LoadID AND ldm.work_type IS NULL),StagingPctFull)  
      else (SELECT ROUND((ISNULL(StagingCubes,0)/StagingCapacity) * 100,0)  
        FROM t_load_master ldm (NOLOCK)   
       WHERE ldm.load_id = LoadID AND ldm.work_type IS NULL AND StagingCapacity > 0)  
      end  
  --*1214428 Shipping Door management - New WW page needs color change v1

UPDATE #report  
   SET Color = CASE WHEN StagingPctFull BETWEEN 1 AND 10 THEN '{{BGCOLOR=#973ADF}}'  
     WHEN StagingPctFull BETWEEN 11 AND 25 THEN '{{BGCOLOR=#FFFF66}}' 
     WHEN StagingPctFull BETWEEN 26 AND 75 THEN '{{BGCOLOR=#66FF66}}'  
     WHEN StagingPctFull BETWEEN 76 AND 90 THEN '{{BGCOLOR=#FFFF66}}'  
     WHEN StagingPctFull >= 91 THEN '{{BGCOLOR=#973ADF}}'  
     WHEN StagingCapacity = 0 AND StagingPctFull IS NOT NULL THEN '{{BGCOLOR=#99CCFF}}'  
     WHEN EXISTS(SELECT 'TRUE'   
          FROM t_load_master ldm (NOLOCK)   
         WHERE ldm.work_type IS NULL   
           AND ldm.status = 'R'   
           AND ldm.load_id = LoadID   
           AND StagingPctFull = 0)  
       THEN '{{BGCOLOR=#973ADF}}'  
     ELSE '{{BGCOLOR=WHITE}}'  
    END  
  --v1
SELECT   
 Color,  
 StagingCapacity,  
 StagingCubes,  
 ID,  
 Door,  
 equipment_id,  
 LoadID as load_id,  
 nbr_of_orders,  
 pct_loaded,  
 pct_picked,  
 UPH_Need,  
 UPH_Upper,  
 upholstery_stage_loc,  
 CG_Need,  
 UPPER_Need,  
 PCS_to_load,  
 StagingPctFull,  
 disp_datetime,  
 [status],  
 current_trailer_loc,  
 zone,  
 wh_id,  
 due,  
 LoadMinRemain   
    ,Non_Fwd_Bld_Qty   
 ,'UPHOLSTERY' AS UPHOLSTERY   
 ,'CASEGOOD' AS CASEGOOD   
 ,'CASEUPPER' AS CASEUPPER   
 ,num_users   
 FROM #report  
ORDER BY Door   
   
  
 DROP TABLE #tempsum  
  
  
ExitLabel:  
  
RETURN
